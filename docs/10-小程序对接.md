# 10 · 小程序对接示例

> 本模块只提供接口与调用示例，不包含正式页面。页面由前端同学负责；把下面的 `request` 封装和调用片段接进现有页面即可。

## 1. 配置

`miniprogram/config.js`：

```js
module.exports = {
  // 开发者工具：本机；真机：电脑局域网 IP；上线：HTTPS 合法域名
  baseUrl: 'http://127.0.0.1:3000',
  // 本地没有 AppID / 无法弹手机号授权时，用 mock 手机号联调
  mockPhone: '13800138000',
};
```

开发者工具：

- 详情 → 本地设置 → 勾选「不校验合法域名、web-view（业务域名）、TLS 版本以及 HTTPS 证书」。
- 真机调试时，手机与电脑同一 Wi-Fi，`baseUrl` 改成电脑局域网 IP。
- 上线前把 `baseUrl` 改成 HTTPS 域名，并在微信公众平台配置 request 合法域名。

## 2. 统一请求封装

`miniprogram/utils/request.js`：

```js
const { baseUrl } = require('../config');

const TOKEN_KEY = 'moto_token';

function getToken() {
  return wx.getStorageSync(TOKEN_KEY) || '';
}

function setToken(token) {
  wx.setStorageSync(TOKEN_KEY, token);
}

function clearToken() {
  wx.removeStorageSync(TOKEN_KEY);
}

/**
 * 统一请求：自动带 token、统一解析信封、统一错误提示。
 * 业务错误（code 1xxx）会 reject，调用方可以按 code 特殊处理。
 */
function request(path, { method = 'GET', data, auth = true, showError = true } = {}) {
  return new Promise((resolve, reject) => {
    const header = { 'Content-Type': 'application/json' };
    if (auth && getToken()) header.Authorization = `Bearer ${getToken()}`;

    wx.request({
      url: `${baseUrl}${path}`,
      method,
      data,
      header,
      success(res) {
        const body = res.data || {};
        if (res.statusCode === 401 || body.code === 401) {
          clearToken();
          if (showError) wx.showToast({ title: '请先登录', icon: 'none' });
          reject({ code: 401, message: '请先登录' });
          return;
        }
        if (body.code === 0) {
          resolve(body.data);
          return;
        }
        if (showError) {
          wx.showToast({ title: body.message || '操作失败', icon: 'none' });
        }
        reject(body);
      },
      fail(error) {
        if (showError) wx.showToast({ title: '网络异常，请重试', icon: 'none' });
        reject({ code: -1, message: '网络异常', error });
      },
    });
  });
}

module.exports = { request, getToken, setToken, clearToken };
```

## 3. 登录

`pages/login/login.js`：

```js
const { request, setToken } = require('../../utils/request');

Page({
  data: {
    loading: false,
    nickname: '',
    avatarUrl: '',
  },

  onChooseAvatar(e) {
    this.setData({ avatarUrl: e.detail.avatarUrl });
  },

  onNicknameInput(e) {
    this.setData({ nickname: e.detail.value });
  },

  async onLogin() {
    if (this.data.loading) return;
    this.setData({ loading: true });
    try {
      const loginRes = await new Promise((resolve, reject) => {
        wx.login({ success: resolve, fail: reject });
      });

      // 本地没有 AppID 时，用 mock: 前缀得到稳定 openid
      const code = loginRes.code || `mock:demo-${Date.now()}`;

      const data = await request('/api/auth/wechat/login', {
        method: 'POST',
        auth: false,
        data: {
          code,
          nickname: this.data.nickname || undefined,
          avatarUrl: this.data.avatarUrl || undefined,
        },
      });
      setToken(data.token);
      wx.setStorageSync('moto_user', data.user);
      wx.showToast({ title: '登录成功', icon: 'success' });
      wx.navigateBack();
    } catch (error) {
      // request 已统一提示；这里只处理业务分支
      if (error.code === 1008) {
        wx.showModal({ title: '账号被禁用', content: '请联系活动管理员', showCancel: false });
      }
    } finally {
      this.setData({ loading: false });
    }
  },
});
```

`login.wxml` 关键按钮：

```xml
<button open-type="chooseAvatar" bindchooseavatar="onChooseAvatar">选择头像</button>
<input type="nickname" bindinput="onNicknameInput" placeholder="请输入昵称" />
<button bindtap="onLogin" loading="{{loading}}">微信登录</button>
```

> 本地无 AppID 时 `wx.login` 仍可能返回 code，但真机无法换 openid。后端 `WECHAT_MOCK=true` 时任意 code 都能生成稳定 openid；想稳定复现某用户，直接传 `mock:demo-user-1`。

## 4. 手机号授权

`pages/activity/detail.wxml`：

```xml
<button
  open-type="getPhoneNumber"
  bindgetphonenumber="onGetPhoneNumber"
  disabled="{{signing}}"
>
  授权手机号并报名
</button>
```

`pages/activity/detail.js`：

```js
const { request } = require('../../utils/request');
const { mockPhone } = require('../../config');

Page({
  data: {
    activity: null,
    signing: false,
  },

  onLoad(options) {
    this.activityId = Number(options.id);
    this.loadDetail();
  },

  async loadDetail() {
    const activity = await request(`/api/activities/${this.activityId}`, { auth: true });
    this.setData({ activity });
  },

  async onGetPhoneNumber(e) {
    // 用户拒绝授权
    if (!e.detail.code) {
      // 本地联调：用 mock 手机号兜底；真实环境应提示用户
      try {
        await request('/api/auth/wechat/phone', {
          method: 'POST',
          data: { phone: mockPhone },
        });
      } catch (error) {
        wx.showToast({ title: '需要手机号才能报名', icon: 'none' });
        return;
      }
      await this.signup();
      return;
    }

    try {
      await request('/api/auth/wechat/phone', {
        method: 'POST',
        data: { code: e.detail.code },
      });
      await this.signup();
    } catch (error) {
      if (error.code === 1011) {
        wx.showToast({ title: '该手机号已绑定其他账号', icon: 'none' });
      }
    }
  },

  async signup() {
    if (this.data.signing) return;
    this.setData({ signing: true });
    try {
      const data = await request(`/api/activities/${this.activityId}/signup`, {
        method: 'POST',
        data: { remark: '' },
      });
      wx.showModal({
        title: '报名成功',
        content: `报名编号：${data.signup.signupNo}`,
        showCancel: false,
      });
      await this.loadDetail();
    } catch (error) {
      // 1002 未开始/已结束；1003 名额满；1004 已报名；1005 已下架；1010 未授权
      const messages = {
        1002: '当前不在报名时间',
        1003: '名额已满，看看其他活动吧',
        1004: '你已经报名过该活动',
        1005: '活动已下架',
        1010: '请先授权手机号',
      };
      if (messages[error.code]) {
        wx.showToast({ title: messages[error.code], icon: 'none' });
      }
    } finally {
      this.setData({ signing: false });
    }
  },
});
```

## 5. 活动列表 / 详情

`pages/activity/list.js`：

```js
const { request } = require('../../utils/request');

Page({
  data: { list: [], page: 1, pageSize: 10, total: 0, loading: false },

  onShow() {
    this.setData({ page: 1, list: [] });
    this.load();
  },

  async load() {
    if (this.data.loading) return;
    this.setData({ loading: true });
    try {
      const data = await request(
        `/api/activities?page=${this.data.page}&pageSize=${this.data.pageSize}`,
        { auth: true },
      );
      this.setData({
        list: this.data.page === 1 ? data.list : this.data.list.concat(data.list),
        total: data.total,
        loading: false,
      });
    } catch (error) {
      this.setData({ loading: false });
    }
  },

  onReachBottom() {
    if (this.data.list.length >= this.data.total) return;
    this.setData({ page: this.data.page + 1 }, () => this.load());
  },

  goDetail(e) {
    wx.navigateTo({ url: `/pages/activity/detail?id=${e.currentTarget.dataset.id}` });
  },
});
```

列表项可直接用返回字段：

```xml
<view wx:for="{{list}}" wx:key="id" bindtap="goDetail" data-id="{{item.id}}">
  <image src="{{item.cover}}" mode="aspectFill" />
  <text>{{item.title}}</text>
  <text>{{item.startAt}}</text>
  <text>剩余 {{item.remaining}} / {{item.quota}}</text>
  <text wx:if="{{item.signedUp}}">已报名</text>
  <text wx:elif="{{!item.signupOpen}}">未开放报名</text>
  <text wx:else>立即报名</text>
</view>
```

> 时间字段是 UTC ISO，页面展示前用 `new Date(item.startAt)` 转本地；不要在模板里直接截字符串。

## 7. 错误码前端处理建议

| code | 前端动作 |
| --- | --- |
| 401 | 清 token，跳登录；登录后回到原页面 |
| 403 | 提示无权限；后台跳回上一页 |
| 1002 | 刷新活动详情，按钮置灰，提示报名时间 |
| 1003 | 按钮置灰，推荐其他活动 |
| 1004 | 展示「已报名」与报名编号，不再请求 |
| 1005 | 返回活动列表 |
| 1008 | 清登录态，弹「账号被禁用」 |
| 1010 | 拉起手机号授权按钮 |
| 1011 | 提示换手机号或联系管理员 |
| 1012 | 刷新详情，重新编辑 |

## 8. 联调顺序

1. 后端：`npm run db:reset && npm run mock -- --clear && npm start`。
2. 小程序：先调 `/api/health` 确认网络通。
3. 登录：`mock:demo-user-1` → 拿 token。
4. 手机号：`phoneCode: mock:13800138000` 或 `phone: 13800138000`。
5. 活动列表 → 详情 → 报名 → 我的报名。
6. 后台「活动管理 → 统计」看报名是否出现；「用户管理」看用户手机号 / 状态。
7. 后台禁用该用户 → 小程序刷新 → 应提示 1008。

## 9. 注意事项

- 不要把 token 放在 URL query；统一放 `Authorization` header。
- 不要在前端判断“名额是否满”作为唯一依据；以服务端返回为准。
- 报名按钮要做防抖，但**服务端仍然必须防重**；前端防抖只是体验优化。
- 图片域名需要配置 download 合法域名；本地演示可用 `/admin/assets/*.svg` 或自己的 CDN。
- 真机调试不能用 `127.0.0.1`，要用电脑局域网 IP；手机与电脑同一 Wi-Fi。
- 上线前把 `WECHAT_MOCK=false`，配置 AppID / Secret，并在微信公众平台配置 request 域名。
